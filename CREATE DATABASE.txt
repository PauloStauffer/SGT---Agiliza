IF OBJECT_ID(N'dbo.usuario', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.usuario (
        id_usuario INT IDENTITY(1,1) PRIMARY KEY,
        nome NVARCHAR(255) NOT NULL,
        email NVARCHAR(255) NOT NULL UNIQUE,
        senha NVARCHAR(255) NOT NULL
    );
END;
GO

IF COL_LENGTH(N'dbo.usuario', N'curso') IS NULL
    ALTER TABLE dbo.usuario ADD curso NVARCHAR(255) NULL;
IF COL_LENGTH(N'dbo.usuario', N'bio') IS NULL
    ALTER TABLE dbo.usuario ADD bio NVARCHAR(MAX) NULL;
IF COL_LENGTH(N'dbo.usuario', N'fuso_horario') IS NULL
    ALTER TABLE dbo.usuario ADD fuso_horario NVARCHAR(100) NOT NULL
        CONSTRAINT DF_usuario_fuso_horario DEFAULT N'America/Sao_Paulo';
IF COL_LENGTH(N'dbo.usuario', N'idioma') IS NULL
    ALTER TABLE dbo.usuario ADD idioma NVARCHAR(10) NOT NULL
        CONSTRAINT DF_usuario_idioma DEFAULT N'pt-BR';
GO

IF OBJECT_ID(N'dbo.area_da_vida', N'U') IS NULL

    CREATE TABLE dbo.area_da_vida (

IF OBJECT_ID(N'dbo.meta', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.meta (
        id_meta_pk INT IDENTITY(1,1) PRIMARY KEY,
        id_usuario_fk INT NOT NULL,
        id_area_fk INT NULL,
        titulo NVARCHAR(255) NOT NULL,
        data_alvo DATE NULL,
        criada_em DATETIME2 NOT NULL CONSTRAINT DF_meta_criada_em DEFAULT SYSUTCDATETIME(),
        CONSTRAINT FK_Meta_Usuario FOREIGN KEY (id_usuario_fk)

    );
END;
GO

IF OBJECT_ID(N'dbo.tarefa', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.tarefa (
        id_tarefa_pk INT IDENTITY(1,1) PRIMARY KEY,
        id_usuario_fk INT NOT NULL,
        id_area_fk INT NULL,
        id_meta_fk INT NULL,
        titulo NVARCHAR(255) NOT NULL,
        descricao NVARCHAR(MAX) NULL,
        statu NVARCHAR(50) NOT NULL CONSTRAINT DF_tarefa_statu DEFAULT N'Não iniciada',
        data_limite DATE NULL,
        horario_inicio TIME NULL,
        prioridade NVARCHAR(50) NULL,
        criticidade NVARCHAR(50) NULL,
        complexidade NVARCHAR(50) NULL,
        duracao_estimada NVARCHAR(50) NULL,
        importante BIT NOT NULL CONSTRAINT DF_tarefa_importante DEFAULT 0,
        adiamentos INT NOT NULL CONSTRAINT DF_tarefa_adiamentos DEFAULT 0,
        tempo_gasto INT NOT NULL CONSTRAINT DF_tarefa_tempo_gasto DEFAULT 0,
        data_conclusao DATETIME2 NULL,
        data_criacao DATETIME2 NOT NULL CONSTRAINT DF_tarefa_data_criacao DEFAULT SYSUTCDATETIME(),
        CONSTRAINT FK_Tarefa_Usuario FOREIGN KEY (id_usuario_fk)
            REFERENCES dbo.usuario(id_usuario),
        CONSTRAINT FK_Tarefa_Area FOREIGN KEY (id_area_fk)
            REFERENCES dbo.area_da_vida(id_area_pk),
        CONSTRAINT FK_Tarefa_Meta FOREIGN KEY (id_meta_fk)
            REFERENCES dbo.meta(id_meta_pk)
    );
END;
GO

IF COL_LENGTH(N'dbo.tarefa', N'id_meta_fk') IS NULL
    ALTER TABLE dbo.tarefa ADD id_meta_fk INT NULL;
IF COL_LENGTH(N'dbo.tarefa', N'horario_inicio') IS NULL

IF COL_LENGTH(N'dbo.tarefa', N'criticidade') IS NULL
    ALTER TABLE dbo.tarefa ADD criticidade NVARCHAR(50) NULL;
IF COL_LENGTH(N'dbo.tarefa', N'importante') IS NULL
    ALTER TABLE dbo.tarefa ADD importante BIT NOT NULL
        CONSTRAINT DF_tarefa_importante DEFAULT 0;
IF COL_LENGTH(N'dbo.tarefa', N'adiamentos') IS NULL
    ALTER TABLE dbo.tarefa ADD adiamentos INT NOT NULL
        CONSTRAINT DF_tarefa_adiamentos DEFAULT 0;
IF COL_LENGTH(N'dbo.tarefa', N'tempo_gasto') IS NULL
    ALTER TABLE dbo.tarefa ADD tempo_gasto INT NOT NULL
        CONSTRAINT DF_tarefa_tempo_gasto DEFAULT 0;
IF COL_LENGTH(N'dbo.tarefa', N'data_conclusao') IS NULL
    ALTER TABLE dbo.tarefa ADD data_conclusao DATETIME2 NULL;
IF COL_LENGTH(N'dbo.tarefa', N'data_criacao') IS NULL
    ALTER TABLE dbo.tarefa ADD data_criacao DATETIME2 NOT NULL
        CONSTRAINT DF_tarefa_data_criacao DEFAULT SYSUTCDATETIME();
